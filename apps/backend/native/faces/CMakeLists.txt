cmake_minimum_required(VERSION 3.24)
project(context_use_face_build LANGUAGES C CXX)
include(ExternalProject)
set(opencv_install ${CMAKE_BINARY_DIR}/opencv)
ExternalProject_Add(opencv
  URL https://github.com/opencv/opencv/archive/refs/tags/4.10.0.tar.gz
  URL_HASH SHA256=b2171af5be6b26f7a06b1229948bbb2bdaa74fcf5cd097e0af6378fce50a6eb9
  CMAKE_ARGS -DCMAKE_INSTALL_PREFIX=${opencv_install} -DCMAKE_BUILD_TYPE=MinSizeRel
    -DBUILD_SHARED_LIBS=OFF -DBUILD_LIST=core,imgproc,imgcodecs,dnn,objdetect
    -DBUILD_TESTS=OFF -DBUILD_PERF_TESTS=OFF -DBUILD_EXAMPLES=OFF -DBUILD_opencv_apps=OFF
    -DBUILD_JAVA=OFF -DBUILD_opencv_python3=OFF -DBUILD_opencv_gapi=OFF -DWITH_ADE=OFF
    -DWITH_IPP=OFF -DWITH_ITT=OFF -DWITH_OPENCL=OFF -DWITH_TBB=OFF -DWITH_OBSENSOR=OFF
    -DWITH_FFMPEG=OFF -DWITH_GSTREAMER=OFF -DWITH_GTK=OFF -DWITH_QT=OFF
    -DWITH_TIFF=OFF -DWITH_OPENEXR=OFF -DWITH_JASPER=OFF -DWITH_JPEG=ON -DWITH_PNG=ON
    -DBUILD_JPEG=ON -DBUILD_PNG=ON -DBUILD_ZLIB=ON -DBUILD_PROTOBUF=ON -DBUILD_WEBP=ON -DWITH_WEBP=ON
    -DCPU_DISPATCH= -DCMAKE_POLICY_VERSION_MINIMUM=3.5
  BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --parallel 4)
ExternalProject_Add(face-analyzer
  DEPENDS opencv
  SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/engine
  BINARY_DIR ${CMAKE_BINARY_DIR}/runtime
  CMAKE_ARGS -DOpenCV_DIR=${opencv_install}/lib/cmake/opencv4
    -DCMAKE_BUILD_TYPE=MinSizeRel -DCMAKE_EXE_LINKER_FLAGS=${CMAKE_EXE_LINKER_FLAGS}
  BUILD_ALWAYS TRUE
  INSTALL_COMMAND "")
